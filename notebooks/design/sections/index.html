beams = []
for i, sec in enumerate(sections):
    ypos = i * 3
    bm = ada.Beam(f"bm_{sec.name}", (0, ypos, 0), (5, ypos, 0), sec=sec)
    beams.append(bm)

a = ada.Assembly() / beams
a.show()
